import os
from datetime import date

import httpx
from fastmcp import FastMCP

mcp = FastMCP("chess-com")

BASE = "https://api.chess.com/pub"
HEADERS = {"User-Agent": "chesscom-mcp/1.0 (contact: gabin.barriere2012@gmail.com }


async def fetch(path: str):
    async with httpx.AsyncClient(headers=HEADERS, timeout=20) as client:
        r = await client.get(f"{BASE}{path}")
        if r.status_code == 404:
            return {"error": "Introuvable : pseudo ou mois invalide"}
        r.raise_for_status()
        return r.json()


@mcp.tool()
async def get_profile(username: str) -> dict:
    """Profil public d'un joueur chess.com (pays, date d'inscription, statut)."""
    return await fetch(f"/player/{username.lower()}")


@mcp.tool()
async def get_stats(username: str) -> dict:
    """Classements et records d'un joueur : rapid, blitz, bullet, puzzles, etc."""
    return await fetch(f"/player/{username.lower()}/stats")


@mcp.tool()
async def get_archives(username: str) -> dict:
    """Liste des mois pour lesquels des parties existent."""
    return await fetch(f"/player/{username.lower()}/games/archives")


@mcp.tool()
async def get_games(
    username: str,
    year: int | None = None,
    month: int | None = None,
    limit: int = 20,
):
    """Parties d'un joueur pour un mois donne (mois courant par defaut).
    Renvoie les `limit` parties les plus recentes avec leur PGN."""
    today = date.today()
    year = year or today.year
    month = month or today.month
    data = await fetch(f"/player/{username.lower()}/games/{year}/{month:02d}")
    if "error" in data:
        return data
    games = data.get("games", [])[-limit:]
    return [
        {
            "url": g.get("url"),
            "time_class": g.get("time_class"),
            "time_control": g.get("time_control"),
            "rated": g.get("rated"),
            "white": g.get("white"),
            "black": g.get("black"),
            "pgn": g.get("pgn"),
        }
        for g in games
    ]


if __name__ == "__main__":
    port = int(os.environ.get("PORT", 8000))
    mcp.run(transport="http", host="0.0.0.0", port=port)
